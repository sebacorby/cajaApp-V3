"use client";

import { useEffect, useMemo, useRef, useState } from "react";
import { Bot, CheckCircle2, CircleDot, FileText, Loader2, Sparkles, TriangleAlert } from "lucide-react";
import {
  CARD_IMPORT_LIVE_EVENT,
  type CardImportLiveEventDetail,
  type CardImportStatus,
} from "@/lib/finance/card-import-api";

type Tone = "info" | "working" | "success" | "error";
type Message = { id: string; title: string; body: string; tone: Tone; timestamp: string };
type Recovery = {
  state?: "stream_error" | "checking_partial" | "partial_valid" | "partial_invalid" | "retrying";
  attempt?: number;
  maxAttempts?: number;
  partialCharacters?: number;
  errorMessage?: string;
  errorRef?: string;
  nextAttempt?: number;
};

const nf = new Intl.NumberFormat("es-AR");

function stageTitle(stage?: string): string | null {
  return ({
    queued: "Importación en cola",
    loading_document: "Cargando documento",
    extracting_raw_text: "Leyendo el PDF",
    sending_raw_text_to_ai: "Enviando el documento a Gemma",
    receiving_ai_stream: "Gemma está procesando el resumen",
    validating_ai_response: "Validando la respuesta del modelo",
    persisting_preview: "Preparando la vista previa",
    preview_ready: "Importación lista",
    failed: "Importación detenida",
  } as Record<string, string>)[stage || ""] ?? null;
}

function toneClass(tone: Tone): string {
  if (tone === "success") return "border-emerald-500/25 bg-emerald-500/8";
  if (tone === "error") return "border-destructive/30 bg-destructive/5";
  if (tone === "working") return "border-primary/25 bg-primary/5";
  return "border-border bg-muted/25";
}

function iconFor(tone: Tone) {
  if (tone === "success") return CheckCircle2;
  if (tone === "error") return TriangleAlert;
  if (tone === "working") return Loader2;
  return CircleDot;
}

function progressBody(status: CardImportStatus): string {
  return status.progress?.message?.trim() || "Procesando…";
}

export function CardImportLiveConsole() {
  const [messages, setMessages] = useState<Message[]>([]);
  const [active, setActive] = useState(false);
  const [draftId, setDraftId] = useState<string | null>(null);
  const scrollRef = useRef<HTMLDivElement>(null);
  const stageRef = useRef<string | null>(null);
  const modelRef = useRef<string | null>(null);
  const rawRef = useRef<string | null>(null);
  const connectionRef = useRef<string | null>(null);
  const lastChunksRef = useRef(0);
  const recoveringRef = useRef(false);
  const attemptRef = useRef(1);
  const recoverySignatureRef = useRef<string | null>(null);

  const append = (message: Message) => {
    setMessages((current) => current.some((item) => item.id === message.id) ? current : [...current, message]);
  };

  const upsert = (message: Message) => {
    setMessages((current) => {
      const index = current.findIndex((item) => item.id === message.id);
      if (index < 0) return [...current, message];
      const next = [...current];
      next[index] = message;
      return next;
    });
  };

  useEffect(() => {
    const onEvent = (event: Event) => {
      const detail = (event as CustomEvent<CardImportLiveEventDetail>).detail;
      if (!detail) return;

      if (detail.type === "reset") {
        stageRef.current = null;
        modelRef.current = null;
        rawRef.current = null;
        connectionRef.current = null;
        lastChunksRef.current = 0;
        recoveringRef.current = false;
        attemptRef.current = 1;
        recoverySignatureRef.current = null;
        setDraftId(null);
        setActive(true);
        setMessages([{ id: "pdf", title: "PDF recibido", body: `${detail.fileName} · preparando la importación`, tone: "info", timestamp: detail.timestamp }]);
        return;
      }

      if (detail.type === "upload_failed") {
        append({ id: "upload-failed", title: "No se pudo iniciar la importación", body: detail.message, tone: "error", timestamp: detail.timestamp });
        setActive(false);
        return;
      }

      if (detail.type === "started") {
        setDraftId(detail.draftId);
        append({ id: `started:${detail.draftId}`, title: "Documento aceptado", body: `${detail.pageCount} páginas detectadas. El backend creó el trabajo de extracción.`, tone: "success", timestamp: detail.timestamp });
        return;
      }

      const { status } = detail;
      setDraftId(detail.draftId);
      const telemetry = status.telemetry;
      const model = [telemetry?.model?.name, telemetry?.model?.provider].filter(Boolean).join(" · ");
      if (model && model !== modelRef.current) {
        modelRef.current = model;
        append({ id: `model:${detail.draftId}`, title: "Modelo asignado", body: model, tone: "info", timestamp: detail.timestamp });
      }

      const raw = telemetry?.rawExtraction;
      if (raw) {
        const signature = `${raw.pageCount}:${raw.characterCount}:${raw.durationMs}`;
        if (signature !== rawRef.current) {
          rawRef.current = signature;
          append({
            id: `raw:${detail.draftId}`,
            title: "Texto del PDF extraído",
            body: [
              typeof raw.pageCount === "number" ? `${nf.format(raw.pageCount)} páginas` : null,
              typeof raw.characterCount === "number" ? `${nf.format(raw.characterCount)} caracteres` : null,
              typeof raw.durationMs === "number" ? `${nf.format(raw.durationMs)} ms` : null,
            ].filter(Boolean).join(" · "),
            tone: "success",
            timestamp: detail.timestamp,
          });
        }
      }

      const stage = status.progress?.stage;
      if (stage && stage !== stageRef.current) {
        stageRef.current = stage;
        const title = stageTitle(stage);
        if (title) append({ id: `stage:${detail.draftId}:${stage}`, title, body: progressBody(status), tone: stage === "failed" ? "error" : stage === "preview_ready" ? "success" : "working", timestamp: detail.timestamp });
      }

      const provider = telemetry?.providerProgress as (typeof telemetry.providerProgress & { attempt?: number; maxAttempts?: number; recovery?: Recovery }) | undefined;
      if (provider) {
        const phase = provider.phase || "streaming";
        const chunks = provider.chunkCount ?? 0;
        const chars = provider.contentCharacters ?? 0;
        const thinking = provider.thinkingCharacters ?? 0;
        const elapsed = provider.elapsedMs ?? 0;
        const recovery = provider.recovery;

        const inferredRecovery = !recovery && phase === "connecting" && (chunks > 0 || chars > 0);
        if (inferredRecovery) {
          recoveringRef.current = true;
          upsert({
            id: `recovery-inferred:${detail.draftId}:${attemptRef.current}`,
            title: "Ollama interrumpió el stream",
            body: `${nf.format(chars)} caracteres y ${nf.format(chunks)} fragmentos conservados · comprobando si el JSON recibido ya es utilizable`,
            tone: "error",
            timestamp: detail.timestamp,
          });
        }

        if (!provider.attempt && phase === "connecting" && chunks === 0 && chars === 0 && (recoveringRef.current || lastChunksRef.current > 0)) {
          attemptRef.current += 1;
          recoveringRef.current = false;
          lastChunksRef.current = 0;
          append({
            id: `retry:${detail.draftId}:${attemptRef.current}`,
            title: `Reintentando con Gemma · intento ${attemptRef.current}/3`,
            body: "La respuesta anterior quedó incompleta. CajaApp inició un nuevo intento automático con el mismo PDF.",
            tone: "working",
            timestamp: detail.timestamp,
          });
        }

        const attempt = provider.attempt ?? attemptRef.current;
        const maxAttempts = provider.maxAttempts ?? 3;
        const connectionSignature = `${attempt}:${phase}:${chunks > 0 || chars > 0 ? "retained" : "fresh"}`;
        if (phase === "connecting" && !inferredRecovery && connectionSignature !== connectionRef.current) {
          connectionRef.current = connectionSignature;
          append({
            id: `connect:${detail.draftId}:${attempt}`,
            title: `Conectando con Ollama · intento ${attempt}/${maxAttempts}`,
            body: attempt > 1 ? "Esperando el primer fragmento del nuevo intento." : "La solicitud fue enviada al modelo y esperamos el primer fragmento del stream.",
            tone: "working",
            timestamp: detail.timestamp,
          });
        }

        if (phase === "streaming" || phase === "completed") {
          const recovered = phase === "completed" && recoveringRef.current;
          upsert({
            id: `stream:${detail.draftId}:${attempt}`,
            title: recovered ? "JSON recuperado correctamente" : phase === "completed" ? `Respuesta del modelo recibida · intento ${attempt}/${maxAttempts}` : `Gemma está respondiendo en vivo · intento ${attempt}/${maxAttempts}`,
            body: [
              `${nf.format(chunks)} fragmentos`,
              `${nf.format(chars)} caracteres de respuesta`,
              thinking > 0 ? `${nf.format(thinking)} caracteres en canal de razonamiento` : "sin canal de razonamiento separado",
              elapsed > 0 ? `${Math.max(1, Math.round(elapsed / 1000))}s acumulados` : null,
              recovered ? "la respuesta conservada era JSON completo y continúa a validación" : null,
            ].filter(Boolean).join(" · "),
            tone: phase === "completed" ? "success" : "working",
            timestamp: detail.timestamp,
          });
          if (chunks > 0) lastChunksRef.current = chunks;
          if (phase === "completed") recoveringRef.current = false;
        }

        if (recovery?.state) {
          const signature = `${recovery.state}:${recovery.attempt}:${recovery.nextAttempt}:${recovery.partialCharacters}:${recovery.errorRef}`;
          if (signature !== recoverySignatureRef.current) {
            recoverySignatureRef.current = signature;
            const partial = typeof recovery.partialCharacters === "number" ? `${nf.format(recovery.partialCharacters)} caracteres conservados` : "respuesta parcial conservada";
            const ref = recovery.errorRef ? ` · ref ${recovery.errorRef}` : "";
            const map: Record<string, { title: string; tone: Tone; body: string }> = {
              stream_error: { title: "Ollama interrumpió el stream", tone: "error", body: `${recovery.errorMessage || "Error del proveedor"}${ref} · ${partial}` },
              checking_partial: { title: "Comprobando la respuesta conservada", tone: "working", body: `${partial} · verificando si contiene un JSON completo` },
              partial_valid: { title: "JSON recuperado correctamente", tone: "success", body: `${partial} · continúa con la validación normal` },
              partial_invalid: { title: "La respuesta quedó incompleta", tone: "working", body: `${partial} · hace falta otro intento` },
              retrying: { title: `Reintentando con Gemma · intento ${recovery.nextAttempt || attempt}/${recovery.maxAttempts || maxAttempts}`, tone: "working", body: "CajaApp conserva el diagnóstico e inicia otro intento automático." },
            };
            const item = map[recovery.state];
            if (item) append({ id: `recovery:${detail.draftId}:${signature}`, ...item, timestamp: detail.timestamp });
          }
        }
      }

      if (status.status === "failed") {
        append({ id: `failed:${detail.draftId}`, title: "La importación falló", body: status.error?.message || "El backend informó un error sin detalle adicional.", tone: "error", timestamp: detail.timestamp });
        setActive(false);
      }
      if (status.status === "preview_ready") {
        append({ id: `ready:${detail.draftId}`, title: "Vista previa lista", body: "La extracción terminó y CajaApp ya puede mostrar el resultado para revisión.", tone: "success", timestamp: detail.timestamp });
        setActive(false);
      }
    };

    window.addEventListener(CARD_IMPORT_LIVE_EVENT, onEvent);
    return () => window.removeEventListener(CARD_IMPORT_LIVE_EVENT, onEvent);
  }, []);

  useEffect(() => {
    const node = scrollRef.current;
    if (node) node.scrollTo({ top: node.scrollHeight, behavior: "smooth" });
  }, [messages]);

  const shortDraft = useMemo(() => draftId ? `${draftId.slice(0, 8)}…` : null, [draftId]);
  if (!messages.length) return null;

  return (
    <section className="overflow-hidden rounded-2xl border bg-card shadow-sm" data-testid="card-import-live-console" aria-live="polite">
      <div className="flex items-center justify-between gap-3 border-b px-4 py-3">
        <div className="flex min-w-0 items-center gap-3">
          <div className="flex size-9 shrink-0 items-center justify-center rounded-full bg-primary/10 text-primary"><Bot className="size-5" /></div>
          <div className="min-w-0">
            <div className="flex flex-wrap items-center gap-2">
              <h2 className="font-semibold text-foreground">Actividad del modelo</h2>
              {active ? <span className="inline-flex items-center gap-1 rounded-full border border-primary/25 bg-primary/10 px-2 py-0.5 text-[11px] font-medium text-primary"><Sparkles className="size-3" />En vivo</span> : null}
            </div>
            <p className="mt-0.5 text-xs text-muted-foreground">Eventos reales del pipeline de importación{shortDraft ? ` · ${shortDraft}` : ""}</p>
          </div>
        </div>
        {active ? <Loader2 className="size-4 animate-spin text-primary" /> : null}
      </div>
      <div ref={scrollRef} className="max-h-[430px] space-y-3 overflow-y-auto p-4">
        {messages.map((message) => {
          const Icon = iconFor(message.tone);
          return <div key={message.id} className="flex items-start gap-3">
            <div className="mt-1 flex size-7 shrink-0 items-center justify-center rounded-full bg-muted text-muted-foreground"><Icon className={`size-4 ${message.tone === "working" ? "animate-pulse" : ""}`} /></div>
            <div className={`min-w-0 flex-1 rounded-xl border px-4 py-3 ${toneClass(message.tone)}`}>
              <div className="flex items-start justify-between gap-3"><p className="text-sm font-semibold text-foreground">{message.title}</p><time className="shrink-0 text-[10px] text-muted-foreground">{new Intl.DateTimeFormat("es-AR", { hour: "2-digit", minute: "2-digit", second: "2-digit" }).format(new Date(message.timestamp))}</time></div>
              <p className="mt-1 break-words text-sm leading-relaxed text-muted-foreground">{message.body}</p>
            </div>
          </div>;
        })}
      </div>
      <div className="flex items-center gap-2 border-t bg-muted/20 px-4 py-2 text-[11px] text-muted-foreground"><FileText className="size-3.5 shrink-0" /><span>Se muestran etapas, métricas, reintentos y recuperación observable; el resultado financiero sigue validándose por separado.</span></div>
    </section>
  );
}
