import { expect, test, type Page, type Route } from "@playwright/test";
import type { BackupItem } from "../src/lib/finance/backup-restore-api";

const HASH = "b".repeat(64);

function backupFixture(
  id: string,
  kind: BackupItem["kind"],
  status: BackupItem["status"],
  fileName: string,
): BackupItem {
  return {
    id,
    fileName,
    kind,
    status,
    sizeBytes: 4096,
    sha256: HASH,
    manifest: {
      format: "cajaapp-backup-v1",
      createdAt: "2026-07-17T12:34:56.789Z",
      application: "CajaApp V3",
      database: {
        entry: "database.sqlite",
        sha256: "a".repeat(64),
        sizeBytes: 2048,
        integrityCheck: "ok",
        foreignKeyViolations: 0,
        tables: ["MonthClose", "BackupArchive"],
        migrations: [
          "20260716233000_add_month_close",
          "20260717001000_add_backup_restore",
        ],
      },
      source: {
        schemaSha256: HASH,
        migrationsSha256: HASH,
        nodeVersion: "v24.18.0",
      },
    },
    createdAt: "2026-07-17T12:34:56.789Z",
    validatedAt:
      status === "validated" ? "2026-07-17T12:35:00.000Z" : null,
    restoredAt:
      status === "restored" ? "2026-07-17T12:40:00.000Z" : null,
  };
}

async function openDesktopSection(page: Page): Promise<void> {
  await page.setViewportSize({ width: 1280, height: 900 });
  await page.goto("/");
  await page.getByRole("button", { name: /^Respaldo$/i }).click();
  await expect(page.getByTestId("backup-restore-section")).toBeVisible();
}

async function openMobileSection(page: Page): Promise<void> {
  await page.setViewportSize({ width: 390, height: 844 });
  await page.goto("/");
  await page.getByRole("button", { name: "Abrir menú" }).click();
  await page.getByRole("button", { name: /^Respaldo$/i }).click();
  await expect(page.getByTestId("backup-restore-section")).toBeVisible();
}

test("Respaldo crea, valida y restaura con backup previo", async ({
  page,
}: {
  page: Page;
}) => {
  const items: BackupItem[] = [];
  let createBody: unknown = null;
  let restoreCalled = false;

  await page.route("**/api/backup-restore**", async (route: Route) => {
    const request = route.request();
    const url = new URL(request.url());

    if (
      request.method() === "POST" &&
      url.pathname === "/api/backup-restore"
    ) {
      createBody = request.postDataJSON();
      const created = backupFixture(
        "backup-1",
        "manual",
        "created",
        "cajaapp-v3-2026-07-17T12-34-56-789Z-prueba.cajaapp-backup",
      );
      items.unshift(created);
      await route.fulfill({
        status: 201,
        contentType: "application/json",
        body: JSON.stringify(created),
      });
      return;
    }

    if (
      request.method() === "POST" &&
      url.pathname.endsWith("/backup-1/validate")
    ) {
      items[0] = {
        ...items[0],
        status: "validated",
        validatedAt: "2026-07-17T12:35:00.000Z",
      };
      await route.fulfill({
        status: 200,
        contentType: "application/json",
        body: JSON.stringify({
          valid: true,
          manifest: items[0].manifest,
          packageSha256: HASH,
        }),
      });
      return;
    }

    if (
      request.method() === "POST" &&
      url.pathname === "/api/backup-restore/restore"
    ) {
      restoreCalled = true;
      const restored = backupFixture(
        "restored-1",
        "restored_upload",
        "restored",
        "cajaapp-v3-2026-07-17T12-40-00-000Z-restored.cajaapp-backup",
      );
      const safety = backupFixture(
        "safety-1",
        "pre_restore",
        "created",
        "cajaapp-v3-2026-07-17T12-39-59-000Z-before-prueba.cajaapp-backup",
      );
      items.unshift(restored, safety);
      await route.fulfill({
        status: 200,
        contentType: "application/json",
        body: JSON.stringify({
          restored: true,
          backup: restored,
          preRestoreBackup: safety,
        }),
      });
      return;
    }

    await route.fulfill({
      status: 200,
      contentType: "application/json",
      body: JSON.stringify({
        items,
        activities: [],
        backupDirectory: "C:\\Users\\Javi\\AppData\\Local\\CajaAppV3\\backups",
      }),
    });
  });

  await openDesktopSection(page);

  const table = page.getByRole("table", {
    name: "Historial de respaldos locales",
  });
  await expect(table).toContainText("Todavía no hay respaldos.");
  await expect(page.getByLabel("Etiqueta opcional")).toBeVisible();
  await expect(page.getByLabel("Seleccionar paquete de respaldo")).toBeVisible();

  await page.getByLabel("Etiqueta opcional").fill("Prueba");
  await page.getByTestId("create-backup").click();
  await expect.poll(() => createBody).toEqual({ label: "Prueba" });
  await expect(
    page.getByRole("status").filter({ hasText: "creado correctamente" }),
  ).toBeVisible();
  await expect(page.getByTestId("backup-row-backup-1")).toContainText(
    "Manual",
  );

  await page.getByTestId("validate-backup-backup-1").click();
  await expect(
    page.getByRole("status").filter({ hasText: "Integridad verificada" }),
  ).toBeVisible();
  await expect(page.getByTestId("backup-row-backup-1")).toContainText(
    "Validado",
  );

  let confirmation = "";
  page.once("dialog", async (dialog) => {
    confirmation = dialog.message();
    await dialog.accept();
  });
  await page.getByLabel("Seleccionar paquete de respaldo").setInputFiles({
    name: "fixture.cajaapp-backup",
    mimeType: "application/octet-stream",
    buffer: Buffer.from("dummy-package"),
  });
  await page.getByTestId("restore-backup").click();
  await expect.poll(() => confirmation).toContain(
    "CajaApp generará primero un respaldo automático",
  );
  await expect.poll(() => restoreCalled).toBe(true);
  await expect(
    page.getByRole("status").filter({ hasText: "Respaldo previo" }),
  ).toBeVisible();
  await expect(page.getByTestId("backup-row-safety-1")).toContainText(
    "Previo a restauración",
  );
  await expect(page.getByTestId("backup-row-restored-1")).toContainText(
    "Paquete restaurado",
  );
});

test("Respaldo recupera errores y mantiene acciones accesibles en mobile", async ({
  page,
}: {
  page: Page;
}) => {
  const item = backupFixture(
    "backup-mobile",
    "manual",
    "created",
    "respaldo-mobile.cajaapp-backup",
  );
  const items: BackupItem[] = [item];
  let listAttempts = 0;
  let restoreCalled = false;
  let dismissedConfirmation = "";
  let acceptedConfirmation = "";

  await page.route("**/api/backup-restore**", async (route: Route) => {
    const request = route.request();
    const url = new URL(request.url());

    if (
      request.method() === "POST" &&
      url.pathname.endsWith("/backup-mobile/validate")
    ) {
      items[0] = {
        ...items[0],
        status: "validated",
        validatedAt: "2026-07-17T12:35:00.000Z",
      };
      await route.fulfill({
        status: 200,
        contentType: "application/json",
        body: JSON.stringify({
          valid: true,
          manifest: items[0].manifest,
          packageSha256: HASH,
        }),
      });
      return;
    }

    if (
      request.method() === "POST" &&
      url.pathname === "/api/backup-restore/restore"
    ) {
      restoreCalled = true;
      const restored = backupFixture(
        "restored-mobile",
        "restored_upload",
        "restored",
        "restored-mobile.cajaapp-backup",
      );
      const safety = backupFixture(
        "safety-mobile",
        "pre_restore",
        "created",
        "safety-mobile.cajaapp-backup",
      );
      items.unshift(restored, safety);
      await route.fulfill({
        status: 200,
        contentType: "application/json",
        body: JSON.stringify({
          restored: true,
          backup: restored,
          preRestoreBackup: safety,
        }),
      });
      return;
    }

    listAttempts += 1;
    if (listAttempts === 1) {
      await route.fulfill({
        status: 503,
        contentType: "application/json",
        body: JSON.stringify({ message: "Backups temporalmente no disponibles" }),
      });
      return;
    }

    await route.fulfill({
      status: 200,
      contentType: "application/json",
      body: JSON.stringify({
        items,
        activities: [],
        backupDirectory: "C:\\Users\\Javi\\AppData\\Local\\CajaAppV3\\backups",
      }),
    });
  });

  await openMobileSection(page);

  await expect(page.getByRole("alert")).toContainText(
    "Backups temporalmente no disponibles",
  );
  const retry = page.getByRole("button", {
    name: "Reintentar carga de respaldos",
  });
  await retry.focus();
  await expect(retry).toBeFocused();
  await page.keyboard.press("Enter");

  const card = page.getByTestId("backup-card-backup-mobile");
  await expect(card).toBeVisible();
  await expect(card).toContainText("Manual");

  const validateButton = page.getByRole("button", {
    name: "Validar respaldo respaldo-mobile.cajaapp-backup",
  });
  await validateButton.focus();
  await expect(validateButton).toBeFocused();
  await page.keyboard.press("Enter");
  await expect(card).toContainText("Validado");
  await expect(
    page.getByRole("button", {
      name: "Descargar respaldo respaldo-mobile.cajaapp-backup",
    }),
  ).toBeVisible();

  const fileInput = page.getByLabel("Seleccionar paquete de respaldo");
  await fileInput.setInputFiles({
    name: "fixture-mobile.cajaapp-backup",
    mimeType: "application/octet-stream",
    buffer: Buffer.from("dummy-package"),
  });
  await expect(
    page.getByText("Archivo seleccionado: fixture-mobile.cajaapp-backup"),
  ).toBeVisible();

  page.once("dialog", async (dialog) => {
    dismissedConfirmation = dialog.message();
    await dialog.dismiss();
  });
  const restoreButton = page.getByRole("button", {
    name: "Restaurar con respaldo previo",
  });
  await restoreButton.focus();
  await expect(restoreButton).toBeFocused();
  await page.keyboard.press("Enter");
  await expect.poll(() => dismissedConfirmation).toContain(
    "reemplazará la base local",
  );
  expect(restoreCalled).toBe(false);

  page.once("dialog", async (dialog) => {
    acceptedConfirmation = dialog.message();
    await dialog.accept();
  });
  await page.keyboard.press("Enter");
  await expect.poll(() => acceptedConfirmation).toContain(
    "respaldo automático",
  );
  await expect.poll(() => restoreCalled).toBe(true);
  await expect(page.getByTestId("backup-card-safety-mobile")).toContainText(
    "Previo a restauración",
  );
});
