"use client";

import {
  useCallback,
  useDeferredValue,
  useEffect,
  useMemo,
  useState,
} from "react";
import {
  AlertTriangle,
  Banknote,
  CheckCircle2,
  ChevronDown,
  ChevronUp,
  Clock3,
  CreditCard,
  ExternalLink,
  FileSearch,
  Files,
  Loader2,
  RefreshCw,
  Search,
  Table2,
  Wrench,
  XCircle,
} from "lucide-react";
import { Badge } from "@/components/ui/badge";
import { Button } from "@/components/ui/button";
import {
  Card,
  CardContent,
  CardHeader,
  CardTitle,
} from "@/components/ui/card";
import { Input } from "@/components/ui/input";
import {
  getImportCenterDetail,
  listImportCenter,
  type ImportCenterItem,
  type ImportCenterKind,
  type ImportCenterResponse,
  type ImportCenterStatus,
} from "@/lib/finance/import-center-api";
import { useFinanceUI } from "@/lib/finance/ui-store";

type KindFilter = "all" | ImportCenterKind;
type StatusFilter = "all" | ImportCenterStatus;

const KIND_LABELS: Record<ImportCenterKind, string> = {
  card_statement: "Resumen de tarjeta",
  salary_receipt: "Recibo de sueldo",
  debit_csv: "Movimientos débito",
};

const STATUS_LABELS: Record<ImportCenterStatus, string> = {
  processing: "Procesando",
  needs_review: "Para revisar",
  accepted: "Aceptada",
  failed: "Con error",
  superseded: "Reemplazada",
  reversed: "Anulada",
  archived: "Archivada",
};

const METADATA_LABELS: Record<string, string> = {
  banco: "Banco",
  marca: "Marca",
  titular: "Titular",
  empleador: "Empleador",
  empleado: "Empleado",
  periodo: "Período",
  neto: "Neto",
  totalPesos: "Total ARS",
  totalDolares: "Total USD",
  ingresoReal: "Ingreso real",
  baseFutura: "Base futura",
  filas: "Filas",
  aceptadas: "Aceptadas",
  omitidas: "Omitidas",
  rechazadas: "Rechazadas",
  delimitador: "Delimitador",
  codificacion: "Codificación",
};

function kindIcon(kind: ImportCenterKind) {
  if (kind === "card_statement") return CreditCard;
  if (kind === "salary_receipt") return Banknote;
  return Table2;
}

function statusClass(status: ImportCenterStatus): string {
  if (status === "accepted") {
    return "border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-300";
  }
  if (status === "failed") {
    return "border-destructive/30 bg-destructive/10 text-destructive";
  }
  if (status === "processing") {
    return "border-blue-500/30 bg-blue-500/10 text-blue-700 dark:text-blue-300";
  }
  if (status === "needs_review") {
    return "border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-300";
  }
  return "border-muted-foreground/20 bg-muted text-muted-foreground";
}

function formatDate(value: string): string {
  return new Intl.DateTimeFormat("es-AR", {
    dateStyle: "medium",
    timeStyle: "short",
    timeZone: "America/Argentina/Tucuman",
  }).format(new Date(value));
}

function formatBytes(value: number | null): string {
  if (value === null) return "No informado";
  if (value < 1024) return `${value} B`;
  if (value < 1024 * 1024) return `${(value / 1024).toFixed(1)} KB`;
  return `${(value / (1024 * 1024)).toFixed(1)} MB`;
}

function displayMetadata(value: string | number | boolean | null): string {
  if (value === null || value === "") return "No informado";
  if (typeof value === "boolean") return value ? "Sí" : "No";
  return String(value);
}

function SummaryCard({
  label,
  value,
  icon: Icon,
  testId,
}: {
  label: string;
  value: number;
  icon: typeof Files;
  testId: string;
}) {
  return (
    <Card className="shadow-sm" data-testid={testId}>
      <CardContent className="flex items-center justify-between p-4">
        <div>
          <p className="text-xs font-medium uppercase tracking-wide text-muted-foreground">
            {label}
          </p>
          <p className="mt-1 text-2xl font-semibold">{value}</p>
        </div>
        <Icon className="size-5 text-primary" aria-hidden="true" />
      </CardContent>
    </Card>
  );
}

export function ImportacionesSection() {
  const setSection = useFinanceUI((state) => state.setSection);
  const [kind, setKind] = useState<KindFilter>("all");
  const [status, setStatus] = useState<StatusFilter>("all");
  const [search, setSearch] = useState("");
  const deferredSearch = useDeferredValue(search);
  const [offset, setOffset] = useState(0);
  const [data, setData] = useState<ImportCenterResponse | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [expandedId, setExpandedId] = useState<string | null>(null);
  const [detail, setDetail] = useState<ImportCenterItem | null>(null);
  const [detailLoading, setDetailLoading] = useState(false);

  const limit = 25;

  const load = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const response = await listImportCenter({
        kind,
        status,
        search: deferredSearch,
        limit,
        offset,
      });
      setData(response);
      if (
        response.items.length === 0 &&
        offset > 0 &&
        response.pagination.total <= offset
      ) {
        setOffset(Math.max(0, offset - limit));
      }
    } catch (loadError) {
      setError(
        loadError instanceof Error
          ? loadError.message
          : "No se pudo cargar el Centro de importaciones.",
      );
    } finally {
      setLoading(false);
    }
  }, [deferredSearch, kind, offset, status]);

  useEffect(() => {
    void load();
  }, [load]);

  const items = data?.items ?? [];
  const summary = data?.summary ?? {
    total: 0,
    processing: 0,
    needsReview: 0,
    accepted: 0,
    failed: 0,
    corrected: 0,
    reversed: 0,
  };

  const rangeLabel = useMemo(() => {
    if (!data || data.pagination.total === 0) return "Sin resultados";
    const start = data.pagination.offset + 1;
    const end = data.pagination.offset + data.items.length;
    return `${start}–${end} de ${data.pagination.total}`;
  }, [data]);

  const resetPage = useCallback(() => {
    setOffset(0);
    setExpandedId(null);
    setDetail(null);
  }, []);

  const toggleDetail = useCallback(async (item: ImportCenterItem) => {
    if (expandedId === item.id) {
      setExpandedId(null);
      setDetail(null);
      return;
    }
    setExpandedId(item.id);
    setDetail(null);
    setDetailLoading(true);
    try {
      setDetail(await getImportCenterDetail(item));
    } catch (detailError) {
      setError(
        detailError instanceof Error
          ? detailError.message
          : "No se pudo cargar la trazabilidad de la importación.",
      );
    } finally {
      setDetailLoading(false);
    }
  }, [expandedId]);

  return (
    <section className="space-y-6" data-testid="import-center-section">
      <div className="flex flex-col gap-4 lg:flex-row lg:items-start lg:justify-between">
        <div>
          <div className="flex items-center gap-2">
            <Files className="size-6 text-primary" aria-hidden="true" />
            <h1 className="text-2xl font-semibold tracking-tight">
              Centro de importaciones
            </h1>
          </div>
          <p className="mt-2 max-w-3xl text-sm text-muted-foreground">
            Historial unificado de documentos procesados por CajaApp. Consultá
            estados, errores, correcciones y el módulo donde cada importación se
            administra.
          </p>
        </div>
        <Button
          variant="outline"
          onClick={() => void load()}
          disabled={loading}
          data-testid="import-center-refresh"
        >
          {loading ? (
            <Loader2 className="mr-2 size-4 animate-spin" />
          ) : (
            <RefreshCw className="mr-2 size-4" />
          )}
          Actualizar
        </Button>
      </div>

      <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-6">
        <SummaryCard label="Total" value={summary.total} icon={Files} testId="import-summary-total" />
        <SummaryCard label="Procesando" value={summary.processing} icon={Clock3} testId="import-summary-processing" />
        <SummaryCard label="Para revisar" value={summary.needsReview} icon={FileSearch} testId="import-summary-review" />
        <SummaryCard label="Aceptadas" value={summary.accepted} icon={CheckCircle2} testId="import-summary-accepted" />
        <SummaryCard label="Con errores" value={summary.failed} icon={XCircle} testId="import-summary-failed" />
        <SummaryCard label="Corregidas" value={summary.corrected} icon={Wrench} testId="import-summary-corrected" />
      </div>

      <Card className="shadow-sm">
        <CardHeader>
          <CardTitle className="text-base">Buscar y filtrar</CardTitle>
        </CardHeader>
        <CardContent className="grid gap-3 md:grid-cols-[minmax(0,1fr)_220px_220px]">
          <div className="relative">
            <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
            <Input
              value={search}
              onChange={(event) => {
                setSearch(event.target.value);
                resetPage();
              }}
              placeholder="Archivo, banco, empleador, período o error"
              className="pl-9"
              aria-label="Buscar importaciones"
              data-testid="import-center-search"
            />
          </div>
          <select
            value={kind}
            onChange={(event) => {
              setKind(event.target.value as KindFilter);
              resetPage();
            }}
            className="h-10 rounded-md border border-input bg-background px-3 text-sm"
            aria-label="Filtrar por tipo"
            data-testid="import-center-kind"
          >
            <option value="all">Todos los tipos</option>
            <option value="card_statement">Resúmenes de tarjeta</option>
            <option value="salary_receipt">Recibos de sueldo</option>
            <option value="debit_csv">Movimientos débito</option>
          </select>
          <select
            value={status}
            onChange={(event) => {
              setStatus(event.target.value as StatusFilter);
              resetPage();
            }}
            className="h-10 rounded-md border border-input bg-background px-3 text-sm"
            aria-label="Filtrar por estado"
            data-testid="import-center-status"
          >
            <option value="all">Todos los estados</option>
            <option value="processing">Procesando</option>
            <option value="needs_review">Para revisar</option>
            <option value="accepted">Aceptadas</option>
            <option value="failed">Con errores</option>
            <option value="superseded">Reemplazadas</option>
            <option value="reversed">Anuladas</option>
            <option value="archived">Archivadas</option>
          </select>
        </CardContent>
      </Card>

      {error ? (
        <div
          className="flex gap-2 rounded-lg border border-destructive/30 bg-destructive/5 px-4 py-3 text-sm text-destructive"
          role="alert"
        >
          <AlertTriangle className="mt-0.5 size-4 shrink-0" />
          <span>{error}</span>
        </div>
      ) : null}

      <Card className="shadow-sm">
        <CardHeader className="flex flex-row items-center justify-between gap-4">
          <div>
            <CardTitle className="text-base">Historial</CardTitle>
            <p className="mt-1 text-sm text-muted-foreground">{rangeLabel}</p>
          </div>
        </CardHeader>
        <CardContent>
          {loading && !data ? (
            <div className="flex min-h-48 items-center justify-center text-sm text-muted-foreground">
              <Loader2 className="mr-2 size-5 animate-spin" />
              Cargando importaciones…
            </div>
          ) : items.length === 0 ? (
            <div className="min-h-48 rounded-xl border border-dashed p-8 text-center">
              <FileSearch className="mx-auto size-9 text-muted-foreground" />
              <p className="mt-3 font-medium">No hay importaciones para mostrar</p>
              <p className="mt-1 text-sm text-muted-foreground">
                Ajustá los filtros o importá documentos desde Tarjetas, Ingresos o Movimientos.
              </p>
            </div>
          ) : (
            <div className="space-y-3" data-testid="import-center-list">
              {items.map((item) => {
                const Icon = kindIcon(item.kind);
                const expanded = expandedId === item.id;
                const currentDetail = expanded ? detail : null;
                return (
                  <article
                    key={item.id}
                    className="rounded-xl border bg-card"
                    data-testid={`import-center-item-${item.kind}`}
                  >
                    <div className="flex flex-col gap-4 p-4 lg:flex-row lg:items-center">
                      <div className="flex min-w-0 flex-1 gap-3">
                        <div className="flex size-10 shrink-0 items-center justify-center rounded-lg bg-primary/10 text-primary">
                          <Icon className="size-5" aria-hidden="true" />
                        </div>
                        <div className="min-w-0">
                          <div className="flex flex-wrap items-center gap-2">
                            <h2 className="truncate font-medium">{item.title}</h2>
                            <Badge variant="outline" className={statusClass(item.status)}>
                              {STATUS_LABELS[item.status]}
                            </Badge>
                            {item.version && item.version > 1 ? (
                              <Badge variant="secondary">Versión {item.version}</Badge>
                            ) : null}
                          </div>
                          <p className="mt-1 truncate text-sm text-muted-foreground">
                            {item.subtitle}
                          </p>
                          <div className="mt-2 flex flex-wrap gap-x-4 gap-y-1 text-xs text-muted-foreground">
                            <span>{KIND_LABELS[item.kind]}</span>
                            {item.periodKey ? <span>Período {item.periodKey}</span> : null}
                            <span>Actualizada {formatDate(item.updatedAt)}</span>
                            {item.correctionCount > 0 ? (
                              <span>{item.correctionCount} corrección(es)</span>
                            ) : null}
                          </div>
                          {item.error ? (
                            <p className="mt-2 flex items-start gap-1.5 text-sm text-destructive">
                              <AlertTriangle className="mt-0.5 size-4 shrink-0" />
                              <span>{item.error.message}</span>
                            </p>
                          ) : item.issues.length > 0 ? (
                            <p className="mt-2 text-sm text-amber-700 dark:text-amber-300">
                              {item.issues[0]}
                              {item.issues.length > 1
                                ? ` · ${item.issues.length - 1} observación(es) más`
                                : ""}
                            </p>
                          ) : null}
                        </div>
                      </div>

                      <div className="flex flex-wrap gap-2">
                        <Button
                          variant="outline"
                          size="sm"
                          onClick={() => void toggleDetail(item)}
                          aria-expanded={expanded}
                        >
                          {expanded ? (
                            <ChevronUp className="mr-2 size-4" />
                          ) : (
                            <ChevronDown className="mr-2 size-4" />
                          )}
                          {expanded ? "Ocultar detalle" : "Ver detalle"}
                        </Button>
                        <Button
                          size="sm"
                          onClick={() => setSection(item.navigation.section)}
                          data-testid={`open-${item.navigation.section}`}
                        >
                          <ExternalLink className="mr-2 size-4" />
                          {item.navigation.label}
                        </Button>
                      </div>
                    </div>

                    {expanded ? (
                      <div className="border-t bg-muted/20 p-4" data-testid="import-center-detail">
                        {detailLoading ? (
                          <div className="flex items-center text-sm text-muted-foreground">
                            <Loader2 className="mr-2 size-4 animate-spin" />
                            Cargando trazabilidad…
                          </div>
                        ) : currentDetail ? (
                          <div className="space-y-4">
                            {currentDetail.error ? (
                              <div className="rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm">
                                <p className="font-medium text-destructive">
                                  {currentDetail.error.message}
                                </p>
                                {currentDetail.error.stage ? (
                                  <p className="mt-1 text-muted-foreground">
                                    Etapa: {currentDetail.error.stage}
                                  </p>
                                ) : null}
                                {currentDetail.error.details.length > 0 ? (
                                  <ul className="mt-2 list-disc space-y-1 pl-5 text-muted-foreground">
                                    {currentDetail.error.details.map((entry) => (
                                      <li key={entry}>{entry}</li>
                                    ))}
                                  </ul>
                                ) : null}
                              </div>
                            ) : null}

                            {currentDetail.issues.length > 0 ? (
                              <div className="rounded-lg border border-amber-500/30 bg-amber-500/5 p-3 text-sm">
                                <p className="font-medium text-amber-700 dark:text-amber-300">
                                  Observaciones
                                </p>
                                <ul className="mt-2 list-disc space-y-1 pl-5 text-muted-foreground">
                                  {currentDetail.issues.map((entry) => (
                                    <li key={entry}>{entry}</li>
                                  ))}
                                </ul>
                              </div>
                            ) : null}

                            <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
                              <DetailValue label="Archivo" value={currentDetail.fileName} />
                              <DetailValue label="Tamaño" value={formatBytes(currentDetail.sizeBytes)} />
                              <DetailValue label="Páginas" value={currentDetail.pageCount === null ? "No aplica" : String(currentDetail.pageCount)} />
                              <DetailValue label="SHA-256" value={`${currentDetail.sha256.slice(0, 12)}…`} mono />
                              {Object.entries(currentDetail.metadata)
                                .filter(([, value]) => value !== null && value !== "")
                                .map(([key, value]) => (
                                  <DetailValue
                                    key={key}
                                    label={METADATA_LABELS[key] ?? key}
                                    value={displayMetadata(value)}
                                  />
                                ))}
                            </div>

                            {currentDetail.ai ? (
                              <div className="rounded-lg border p-3 text-sm">
                                <p className="font-medium">Extracción IA</p>
                                <p className="mt-1 text-muted-foreground">
                                  {currentDetail.ai.provider} · {currentDetail.ai.model} · {currentDetail.ai.status}
                                </p>
                                {currentDetail.ai.warnings.length > 0 ? (
                                  <p className="mt-2 text-muted-foreground">
                                    {currentDetail.ai.warnings.length} advertencia(s) de validación.
                                  </p>
                                ) : null}
                              </div>
                            ) : (
                              <div className="rounded-lg border p-3 text-sm text-muted-foreground">
                                Esta importación no utilizó IA.
                              </div>
                            )}
                          </div>
                        ) : null}
                      </div>
                    ) : null}
                  </article>
                );
              })}
            </div>
          )}

          <div className="mt-4 flex items-center justify-between gap-3">
            <Button
              variant="outline"
              size="sm"
              disabled={offset === 0 || loading}
              onClick={() => setOffset(Math.max(0, offset - limit))}
            >
              Anterior
            </Button>
            <span className="text-xs text-muted-foreground">{rangeLabel}</span>
            <Button
              variant="outline"
              size="sm"
              disabled={!data?.pagination.hasMore || loading}
              onClick={() => setOffset(offset + limit)}
            >
              Siguiente
            </Button>
          </div>
        </CardContent>
      </Card>
    </section>
  );
}

function DetailValue({
  label,
  value,
  mono = false,
}: {
  label: string;
  value: string;
  mono?: boolean;
}) {
  return (
    <div className="rounded-lg border bg-background p-3">
      <p className="text-xs font-medium uppercase tracking-wide text-muted-foreground">
        {label}
      </p>
      <p className={`mt-1 break-words text-sm ${mono ? "font-mono" : ""}`}>
        {value}
      </p>
    </div>
  );
}
