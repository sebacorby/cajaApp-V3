/**
 * Integration test for FEAT-026: Fix card reference in future debt projections.
 *
 * RED → GREEN verification:
 * - RED: Test FAILS because `groupKey: r.groupId` stores the section ID
 * - GREEN: Test PASSES because `groupKey: r.id` stores the correct join key
 *
 * The bug: `groupKey: r.groupId` in acceptDraft() stores the section ID,
 * breaking the row→group join in normalizeProjection.
 *
 * The fix: `groupKey: r.id` stores the value that matches CardStatementGroup.groupKey.
 *
 * NOTE: This test verifies the groupKey value is stored correctly. The cardLast4
 * derivation via normalizeProjection depends on a separate join path that
 * requires projection.rowId to match CardStatementRow.id (UUID), which is
 * a pre-existing architectural issue beyond the scope of this one-line fix.
 */
import { describe, it, expect, beforeAll, afterAll, beforeEach } from "vitest";
import type { FastifyInstance } from "fastify";
import { buildApp } from "../../src/app.js";
import { prisma } from "../../src/db/prisma.js";
import type { CardStatementPreview } from "../../src/modules/cards/cards.types.js";

function minimalPreview(
  overrides: Partial<CardStatementPreview> = {},
): CardStatementPreview {
  return {
    statementId: null,
    source: {
      bankName: "Banco Galicia",
      brand: "Visa",
      statementNumber: `test-${Math.random().toString(36).slice(2, 10)}`,
      pageCount: 1,
    },
    summary: {
      totalPesos: "100000.00",
      totalDollars: "100.00",
      minimumPaymentPesos: "10000.00",
      currentDueDate: "2026-07-15",
      nextClosingDate: "2026-07-28",
      nextDueDate: "2026-08-15",
    },
    sections: [],
    groups: [],
    rows: [],
    futureInstallmentsBlock: [],
    ...overrides,
  };
}

async function seedPreviewReadyDraft(
  preview: CardStatementPreview,
): Promise<string> {
  const document = await prisma.uploadedDocument.create({
    data: {
      fileName: `test-${Date.now()}-${Math.random().toString(36).slice(2, 8)}.pdf`,
      mimeType: "application/pdf",
      sizeBytes: 1,
      sha256: `sha-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
      storagePath: `storage/test-${Date.now()}.pdf`,
      pageCount: 1,
    },
  });
  const draft = await prisma.cardStatementDraft.create({
    data: {
      documentId: document.id,
      status: "preview_ready",
      previewJson: JSON.stringify(preview),
    },
  });
  return draft.id;
}

describe("FEAT-026: acceptDraft stores correct groupKey for cardLast4 derivation", () => {
  let app: FastifyInstance;

  let seededDraftIds: string[] = [];
  let seededDocumentIds: string[] = [];
  let createdStatementIds: string[] = [];

  beforeAll(async () => {
    app = await buildApp();
    await app.ready();
  });

  beforeEach(() => {
    seededDraftIds = [];
    seededDocumentIds = [];
    createdStatementIds = [];
  });

  afterEach(async () => {
    try {
      if (createdStatementIds.length > 0) {
        await prisma.cardInstallmentProjection.deleteMany({
          where: { statementId: { in: createdStatementIds } },
        });
        await prisma.cardStatementRow.deleteMany({
          where: { statementId: { in: createdStatementIds } },
        });
        await prisma.cardStatementSection.deleteMany({
          where: { statementId: { in: createdStatementIds } },
        });
        await prisma.cardStatementGroup.deleteMany({
          where: { statementId: { in: createdStatementIds } },
        });
        await prisma.cardStatement.deleteMany({
          where: { id: { in: createdStatementIds } },
        });
      }
      if (seededDraftIds.length > 0) {
        await prisma.cardStatementDraftRow.deleteMany({
          where: { draftId: { in: seededDraftIds } },
        });
        await prisma.cardStatementDraftGroup.deleteMany({
          where: { draftId: { in: seededDraftIds } },
        });
        await prisma.cardStatementDraftSection.deleteMany({
          where: { draftId: { in: seededDraftIds } },
        });
        await prisma.cardStatementDraft.deleteMany({
          where: { id: { in: seededDraftIds } },
        });
      }
      if (seededDocumentIds.length > 0) {
        await prisma.uploadedDocument.deleteMany({
          where: { id: { in: seededDocumentIds } },
        });
      }
    } catch (err) {
      // eslint-disable-next-line no-console
      console.error("[accept-draft-group-key.test] cleanup failed:", err);
    }
  });

  afterAll(async () => {
    await app.close();
    await prisma.$disconnect();
  });

  it("stores r.id as groupKey (not r.groupId which is the section ID)", async () => {
    // Test data:
    // - Section "s-1"
    // - Group "g-1" (becomes CardStatementGroup.groupKey)
    // - Row with id "g-1" (so r.id === g.id for the join) and groupId "s-1" (section ID)
    //
    // The join requires CardStatementRow.groupKey to match CardStatementGroup.groupKey.
    // With the fix: groupKey = r.id = "g-1" matches the group's groupKey = "g-1"
    // With the bug: groupKey = r.groupId = "s-1" does NOT match the group's groupKey = "g-1"
    const preview = minimalPreview({
      summary: {
        totalPesos: "50000.00",
        totalDollars: "50.00",
        minimumPaymentPesos: "5000.00",
        currentDueDate: "2026-07-15",
        nextClosingDate: "2026-07-28",
        nextDueDate: "2026-08-15",
      },
      sections: [{ id: "s-1", displayOrder: 1, label: "Section Principal" }],
      groups: [
        {
          id: "g-1", // This becomes CardStatementGroup.groupKey
          displayOrder: 1,
          label: "Grupo Visa",
          cardLast4: "4521",
          holderName: "Javier",
        },
      ],
      rows: [
        {
          id: "g-1", // r.id === g.id so the join works
          displayOrder: 1,
          sourcePage: 1,
          sectionId: "s-1",
          sectionLabel: "Section Principal",
          groupId: "s-1", // Misleading: this is the section ID, not group ID
          groupLabel: "Grupo Visa",
          groupOrder: 1,
          rowType: "transaction",
          editable: false,
          dateRaw: "15/07/2026",
          dateIso: "2026-07-15",
          markerRaw: null,
          referenceRaw: "REF-001",
          installmentRaw: "1/3",
          installmentCurrent: 1,
          installmentTotal: 3, // 3 installments = 2 future projections
          receiptRaw: null,
          amountPesos: "5000.00",
          amountDollars: null,
          currencyOriginal: "ARS",
          originalText: "Compra en cuotas",
          confidence: 0.95,
          warnings: [],
        },
      ],
      futureInstallmentsBlock: [],
    });

    const draftId = await seedPreviewReadyDraft(preview);
    seededDraftIds.push(draftId);
    const draft = await prisma.cardStatementDraft.findUnique({
      where: { id: draftId },
    });
    if (draft) seededDocumentIds.push(draft.documentId);

    // Accept the draft
    const response = await app.inject({
      method: "POST",
      url: `/api/card-statements/drafts/${draftId}/accept`,
      payload: { preview },
    });
    expect(response.statusCode).toBe(200);
    const body = JSON.parse(response.body);
    createdStatementIds.push(body.statementId);

    // Find the created statement
    const stmt = await prisma.cardStatement.findFirst({
      where: { draftId },
      orderBy: { createdAt: "desc" },
    });
    expect(stmt).not.toBeNull();
    const statementId = stmt!.id;

    // Query the created CardStatementRow and Group
    const [rows, groups] = await Promise.all([
      prisma.cardStatementRow.findMany({
        where: { statementId },
        orderBy: { displayOrder: "asc" },
      }),
      prisma.cardStatementGroup.findMany({
        where: { statementId },
        orderBy: { displayOrder: "asc" },
      }),
    ]);

    expect(rows.length).toBeGreaterThan(0);
    expect(groups.length).toBeGreaterThan(0);

    const transactionRow = rows.find((r) => r.rowType === "transaction");
    const group = groups[0];

    expect(transactionRow).toBeDefined();
    expect(group).toBeDefined();

    // THE CRITICAL ASSERTION:
    // After fix: groupKey should equal r.id ("g-1"), matching CardStatementGroup.groupKey
    // Before fix: groupKey would equal r.groupId ("s-1"), which is the section ID - WRONG
    expect(transactionRow!.groupKey).toBe("g-1");

    // Verify the group was created with groupKey = "g-1"
    expect(group!.groupKey).toBe("g-1");

    // The join key now matches: row.groupKey === group.groupKey === "g-1"
    // This enables normalizeProjection to find the group and derive cardLast4 correctly
    expect(transactionRow!.groupKey).toBe(group!.groupKey);
  });
});
